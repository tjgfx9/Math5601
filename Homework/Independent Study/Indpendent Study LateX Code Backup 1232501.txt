\section{Algorithm Analysis: Adaptive Numerical Integration}

\subsection*{General Framework}
Branching, bounding, stopping.

\subsection*{Initialization}
\begin{enumerate}
    \setlength\itemsep{0.2em}
    \item Initialize sum accumulator \( X \gets 0 \)
    \item Initialize iteration count \( k \gets 0 \)
    \item Initialize good intervals \( \text{goodIntervals} \gets [\,] \)
    \item Initialize test intervals \( \text{testIntervals} \gets [a, b] \)
    \item Define total tolerance \( \epsilon \)
    \item Define smallest bound \( \delta \gets \max(\epsilon / 10^k, h_{\min}) \)
\end{enumerate}

\subsection*{Adaptive Integration Algorithm}
\begin{algorithm}[H]
\small
\begin{algorithmic}
\While{$\text{testIntervals}$ not empty}
    \State $(\text{left}, \text{right}) \gets$ first interval in $\text{testIntervals}$
    \State $\text{mid} \gets (\text{left} + \text{right}) / 2$
    \State $\text{S}_1 \gets$ SIMPLE($[\text{left}, \text{mid}]$)
    \State $\text{S}_2 \gets$ SIMPLE($[\text{mid}, \text{right}]$)
    \State $\text{S}  \gets$ SIMPLE($[\text{left}, \text{right}]$)
    \State $k \gets k + 1$
    \State $\epsilon_k \gets \epsilon \cdot \frac{\text{right}-\text{left}}{b-a}$
    \If{$|\text{S}_1+\text{S}_2-\text{S}| \le 15\epsilon_k$}
        \State Add interval to $\text{goodIntervals}$
        \State $X \gets X + \text{S}_1 + \text{S}_2$
        \State Remove interval from $\text{testIntervals}$
    \Else
        \State $h \gets \text{right}-\text{left}$
        \If{$h > \delta$}
            \State Replace $[\text{left},\text{right}]$ with $[\text{left},\text{mid}]$ and $[\text{mid},\text{right}]$
        \Else
            \State $X \gets $X + \text{S}$
            \State Remove interval (warn: tolerance not met)
        \EndIf
    \EndIf
\EndWhile
\State \Return $X$
\end{algorithmic}
\end{algorithm}
